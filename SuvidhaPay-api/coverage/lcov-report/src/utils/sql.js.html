const vehicleBalanceJoin = `
  LEFT JOIN (
    SELECT
      vehicle_id,
      COALESCE(SUM(amount) FILTER (WHERE status <> 'cancelled'), 0) AS total_due,
      COALESCE(SUM(paid_amount) FILTER (WHERE status <> 'cancelled'), 0) AS total_paid
    FROM dues
    GROUP BY vehicle_id
  ) balances ON balances.vehicle_id = v.id
`;
